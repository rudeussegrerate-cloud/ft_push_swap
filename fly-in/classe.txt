zone(abstract):
    attribue = type, position, capacite
    comportement ou role = recoi le drone
otherzonespecifique:
    redefinition
connexion:
    attribue = capacite
    comportement = lie deux zone, permet de faire traverser les drone
drone:
    attribue =  id, nom, action
    comportement = il connais son chemain et vole juste vers la zone ou il doit aller

reseau complet: 
    attribue = zone, connexion
    comportement = forme le reseau complet


Police:
    role = gere les conflit entre les drone sur une zone, ou les max drone
        utilise table.
Parsing:
    attribue = nombre de drone, les zone(depart, hub et fin), metadonnees, connexion
    comportement ou role: il recuper tout les donner du fichier de configuration pour les assigner ensuite
    au class correspondant.

Pathfinding:
    role: trouve le chemain le plus cours pour un drone.

table:
    role = table de réservation partagée (occupancy des zones/connexions dans le temps)


simulation:
    role: ordonnancer, appeler Pathfinding, committer dans table, ET produire la sortie.
